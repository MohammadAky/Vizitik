      ساختار جدول:

      📦 همه محصولات

      لیست کامل محصولات با قیمت‌ها

      SELECT * FROM Product ORDER BY brand, name

      📊 محصولات پرفروش

      محصولات به تعداد سفارش

      SELECT p.name, p.brand, COUNT(oi.id) as orderCount FROM Product p LEFT JOIN OrderItem oi ON p.id = oi.productId GROUP BY p.id ORDER BY orderCount DESC

      📋 سفارشات اخیر

      ۱۰۰ سفارش آخر

      SELECT o.*, c.name as customerName, u.firstName as visitorName FROM Order o JOIN Customer c ON o.customerId = c.id JOIN User u ON o.visitorId = u.id ORDER BY o.createdAt DESC LIMIT 100

      ⏳ سفارشات در انتظار

      سفارشات تایید نشده

      SELECT * FROM Order WHERE status = 'DRAFT' ORDER BY createdAt DESC

      💰 بدهکاران

      مشتریان با مانده حساب منفی

      SELECT c.name, c.phone, le.balanceAfter as debt FROM Customer c JOIN CustomerLedger le ON c.id = le.customerId WHERE le.balanceAfter > 0 AND le.id = (SELECT MAX(id) FROM CustomerLedger WHERE customerId = c.id) ORDER BY debt DESC

      💳 خلاصه پرداخت‌ها

      پرداخت‌ها به ت.method

      SELECT method, COUNT(*) as count, SUM(amount) as total FROM Payment GROUP BY method

      🏃 عملکرد ویزیتورها

      تعداد سفارش و فروش هر ویزیتور

      SELECT u.firstName, u.lastName, COUNT(o.id) as orderCount, SUM(o.finalAmount) as totalSales FROM User u LEFT JOIN Order o ON u.id = o.visitorId WHERE u.role = 'VISITOR' GROUP BY u.id ORDER BY totalSales DESC

      🚚 موجودی ون

      موجودی فعلی ون‌ها

      SELECT u.firstName, u.lastName, p.name as productName, vi.quantityCartons, vi.quantityUnits FROM VanInventory vi JOIN User u ON vi.userId = u.id JOIN Product p ON vi.productId = p.id ORDER BY u.firstName, p.name

      📈 درآمد ماهانه

      درآمد هر ماه شمسی

      SELECT DATE_FORMAT(orderDate, '%Y-%m') as month, COUNT(*) as orderCount, SUM(finalAmount) as revenue FROM Order WHERE status != 'CANCELLED' GROUP BY month ORDER BY month DESC

      📊 آمار دیتابیس

      تعداد ردیف هر جدول

      SELECT TABLE_NAME as tableName, TABLE_ROWS as rowCount FROM information_schema.TABLES WHERE TABLE_SCHEMA = DATABASE() ORDER BY TABLE_ROWS DESC
